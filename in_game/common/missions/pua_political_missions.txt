mission_claim_mcdonalds = {
		icon = mamluk_soldiers
		requires = { mission_escape_from_straszecin }

		bypass = {
			OR = {
				AND = {
					NOT = { country_exists = c:MCD }
					NOT = { country_exists = c:MCB }
				}
				c:MCD = {
					is_subject_of = root
				}
				c:MCB = {
					is_subject_of = root
				}
				AND = {
					country_exists = c:MCD
					c:MCD = {
						num_locations = 0
					}
				}
			}
		}

		enabled = {
			tag = PUA
			NOT = { has_variable = pua_foreign_claim_mission_active }
			NOT = { has_country_modifier = recent_foreign_claims }
			NOT = {
				has_casus_belli_of_type_on = {
					target = c:MCD
					type = casus_belli:cb_annex
				}
			}
			NOT = {
				has_casus_belli_of_type_on = {
					target = c:MCB
					type = casus_belli:cb_annex
				}
			}
		}

		duration = 180

		on_persistent_start = {
			set_variable = pua_foreign_claim_mission_active
		}

		on_persistent_completion = {
			remove_variable = pua_foreign_claim_mission_active
			add_country_modifier = {
				modifier = recent_foreign_claims
				years = 3
				mode = replace
			}
			mcd_save_active_country_as = yes
			if = {
				limit = {
					exists = scope:active_mcd_country
				}
				add_casus_belli = {
					target = scope:active_mcd_country
					type = casus_belli:cb_annex
					years = 20
				}
			}
		}

		on_bypass = {
			remove_variable = pua_foreign_claim_mission_active
		}
	}
