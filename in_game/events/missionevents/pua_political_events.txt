pua_political_events.1 = {
	type = country_event
	title = pua_political_events.1.title
	desc = pua_political_events.1.desc

	immediate = {
		save_scope_as = target_country2
		c:TVI = {
			save_scope_as = target_country
		}
		location:mielec = {
			province_definition = {
				save_scope_as = target_province_definition
			}
		}
	}

	option = {

pua_political_events.2 = {
	type = country_event
	title = pua_political_events.2.title
	desc = pua_political_events.2.desc

	option = {
		name = pua_political_events.2.a
		historical_option = yes
		trigger = {
			scope:target_country2 = {
				can_pay_price = price:pua_buy_twierdza_mielec_price
			}
		}

		scope:target_country2 = {
			set_variable = {
				name = pua_twierdza_purchase_price
				value = price:pua_buy_twierdza_mielec_price
			}
			pay_price = price:pua_buy_twierdza_mielec_price
		}
		add_gold = scope:target_country2.var:pua_twierdza_purchase_price
		scope:target_province_definition = {
			every_location_in_province_definition = {
				limit = {
					owner ?= scope:target_country
					is_ownable = yes
				}
				change_location_owner = scope:target_country2
				change_location_controller = scope:target_country2
			}
		}
		scope:target_country2 = {
			remove_variable = pua_twierdza_purchase_price
			trigger_event_non_silently = pua_political_events.3
		}
	}

	option = {
		name = pua_political_events.2.b
		trigger = {
			is_ai = no
		}

		scope:target_country2 = {
			add_casus_belli = {
				target = root
				type = casus_belli:cb_conquer_province
				province = location:mielec.province
				years = 20
			}
			trigger_event_non_silently = pua_political_events.4
		}
	}
}

pua_political_events.6 = {
	type = country_event
	title = pua_political_events.6.title
	desc = pua_political_events.6.desc

	option = {
		name = pua_political_events.6.a
		historical_option = yes
		if = {
			limit = {
				NOT = {
					is_allied_with = {
						target = scope:target_country3
					}
				}
			}
			create_alliance = {
				target = scope:target_country3
			}
		}
		if = {
			limit = {
				NOT = { country_exists = c:CKZ }
			}
			scope:target_country3 = {
				trigger_event_non_silently = pua_political_events.8
			}
		}
		else_if = {
			limit = {
				NOT = {
					OR = {
						location:ryglice = { owner ?= c:CKZ }
						location:rzepiennik_strzyżewski = { owner ?= c:CKZ }
						location:szerzyny = { owner ?= c:CKZ }
						location:bobowa = { owner ?= c:CKZ }
						location:lużna = { owner ?= c:CKZ }
						location:moszczenica = { owner ?= c:CKZ }
						location:biecz = { owner ?= c:CKZ }
						location:gorlice = { owner ?= c:CKZ }
						location:ropa = { owner ?= c:CKZ }
						location:sękowa = { owner ?= c:CKZ }
						location:uście_gorlickie = { owner ?= c:CKZ }
					}
				}
			}
			scope:target_country3 = {
				trigger_event_non_silently = pua_political_events.8
			}
		}
		else = {
			scope:target_country2 = {
				trigger_event_non_silently = pua_political_events.7
			}
		}
	}

	option = {
		name = pua_political_events.6.b
		trigger = {
			is_ai = no
		}

		scope:target_country3 = {
			trigger_event_non_silently = pua_political_events.9
		}
	}
}
